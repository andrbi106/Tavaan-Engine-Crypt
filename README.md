## استفاده

```php
use Tavaan\Engine\Crypt\Encrypter;
use Tavaan\Engine\Crypt\KeyRing;

// Setup
$key = KeyRing::generateKey();  // 32 random bytes
$keyRing = new KeyRing($key, 1);  // version 1
$crypt = new Encrypter($keyRing);

// Encrypt/Decrypt
$cipher = $crypt->encrypt('متن محرمانه');
$plain = $crypt->decrypt($cipher);

// JSON
$cipher = $crypt->encryptJson(['user_id' => 42, 'email' => 'user@example.com']);
$data = $crypt->decryptJson($cipher);

// AAD (Additional Authenticated Data)
$cipher = $crypt->encrypt('payload', 'context_aad');
$plain = $crypt->decrypt($cipher, 'context_aad');

// HMAC
$mac = $crypt->hmac('data', 'context');

// Key Derivation (HKDF-SHA256)
$derivedKey = $crypt->deriveKey('master', 'context', length: 32);

// Key Rotation
$keyRing->rotate(KeyRing::generateKey());  // version becomes 2, old key kept

// Digital Signatures (RSA)
$keys = $crypt->generateKeyPair();
$sig = $crypt->signWithRsa('payload', $keys['private']);
$valid = $crypt->verifyRsa('payload', $sig, $keys['public']);
```
